# A gobrun --script file (tools/runner.cpp). Goblin 0 is the picker, 1 the
# fighter, 2 the wizard. `walk/act/pick/use X Y` act at screen position X,Y
# (where the cursor's hot spot lands), `exit` clicks the exit sign once it
# shows and goes on to the next level. The goblins find paths as the
# original does (direct lines, or along the level's way points), so some
# walks go in steps.
room AVT003
hint kick
goblin 1
act 186 63      # the fighter kicks the gate: the horn falls
hint horn
goblin 0
pick 163 104    # the picker picks up the horn
hint blow
use 99 93       # and blows it: a branch falls
hint pickaxe
goblin 2
act 90 129      # the wizard turns the branch into a pickaxe
hint take
goblin 0
pick 92 130     # the picker swaps the horn for the pickaxe
hint end
exit
