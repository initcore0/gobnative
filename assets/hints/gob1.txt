# gob2native - hints for Gobliiins (engine/hints.h)
# SPDX-License-Identifier: GPL-3.0-or-later
#
# One block per level, its steps in the order the walkthrough solves them
# (tests/gob1/*.txt, whose `hint ID` lines check this file). The step hinted
# is the one after the last step that is done, so a step's `done` must stay
# true (or a later step's become true) once the player is past it.
#
#   level SCRIPT Title
#   step ID
#     done COND[, COND...]   done when all hold; several done lines: any one
#     goblin N               0 the picker, 1 the fighter, 2 the wizard
#     action walk|act|pick|use
#     at X Y                 where to click (screen pixels; highlighted)
#     item ID                or the item on the map to act on (highlighted)
#     needs ID               an earlier step this one can't be done without
#                            (hinted instead while it isn't done)
#     nudge TEXT             a gentle push
#     pointer TEXT           who, and on what
#     solve TEXT             exactly what to do
#
# Conditions: obj N state S, obj N type T (0 shown, 1 hidden), pocket ID
# (the item id the picker carries), item ID gone|present, item ID cells N
# (the map cells holding it), var SLOT V, var SLOT LO HI, exit (the exit
# sign shows). Find them with gobrun --state-diff.
